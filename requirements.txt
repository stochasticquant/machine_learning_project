# NOTES
#
# 1. Conda alternative. This project was developed in a conda environment.
#    To recreate it that way instead:
#
#       conda create -n laei python=3.12 -y
#       conda activate laei
#       conda install -y pandas numpy scipy scikit-learn matplotlib seaborn \
#                        openpyxl jupyterlab nbconvert requests
#       pip install python-pptx tabulate
#
# 2. Why the pins matter. Two API differences bite if you drop to older
#    versions of the libraries this was built against:
#      - matplotlib >= 3.9 renamed boxplot(labels=) to boxplot(tick_labels=)
#      - pandas 3.x loads string columns as `str` dtype, not `object`
#    Neither changes any result, but code written for one will warn or fail
#    on the other.
#
# 3. Reproducibility. Every split, model and clustering run in this project
#    sets random_state=42. With these pinned versions the numbers in
#    reports/Task3_Task4_Report.md reproduce exactly. Minor-version drift in
#    scikit-learn can shift the last decimal place without being a bug.
#
# 4. Optional extras, deliberately not pinned here because nothing in the
#    committed notebooks imports them:
#      geopandas, shapely, pyproj, folium   # nicer maps than the scatter
#                                           # plots used in notebooks 02/04
#      xgboost, lightgbm                    # further gradient-boosting
#                                           # benchmarks alongside
#                                           # HistGradientBoosting
#
# 5. Not a dependency: `src/laei.py` is this project's own shared module.
#    Import it with sys.path.append("../src"), not pip.
